cmake_minimum_required(VERSION 3.20)
project(MazeRouter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_PYTHON "Build pybind11 extension" OFF)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(router_core src/functions.cpp)
set_target_properties(router_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(router_core PUBLIC src)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(router_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(maze-router src/main.cpp)
target_link_libraries(maze-router PRIVATE router_core)
if(BUILD_PYTHON)
  set(PYBIND11_FINDPYTHON ON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(routing src/binding.cpp)
  target_link_libraries(routing PRIVATE router_core)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python COMPONENTS Interpreter REQUIRED)
  add_test(NAME routing-regressions COMMAND ${Python_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_router.py $<TARGET_FILE:maze-router>)
endif()
